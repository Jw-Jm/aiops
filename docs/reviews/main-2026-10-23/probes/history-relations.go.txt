package resourcestore

import (
 "context"
 "strings"
 "testing"
 "time"
 "github.com/jackc/pgx/v5"
 "k8s.io/apimachinery/pkg/apis/meta/v1/unstructured"
)

type reviewRelationTx struct { pgx.Tx; ids []string; query string }
func (tx *reviewRelationTx) Query(_ context.Context, query string, _ ...any) (pgx.Rows,error) {
 tx.query=query
 return &inventoryRows{ids:tx.ids},nil
}
func TestReviewHistoryRelationsDisappearAcrossClusters(t *testing.T) {
 tenant := "11111111-1111-4111-8111-111111111111"
 pod := "k8s+v1://"+tenant+"/cluster-a/core/Pod/pod-a"
 nodeA := "k8s+v1://"+tenant+"/cluster-a/core/Node/node-a"
 nodeB := "k8s+v1://"+tenant+"/cluster-b/core/Node/node-b"
 obj := unstructured.Unstructured{Object:map[string]any{"spec":map[string]any{"nodeName":"worker-1"}}}
 tx := &reviewRelationTx{ids:[]string{nodeA}}
 got,err:=observedRelations(context.Background(),tx,tenant,pod,"source-a",obj,time.Now())
 if err!=nil || len(got)!=1 {t.Fatalf("single cluster: %v %v",got,err)}
 tx.ids=[]string{nodeA,nodeB}
 got,err=observedRelations(context.Background(),tx,tenant,pod,"source-a",obj,time.Now())
 if err!=nil || len(got)!=0 || strings.Contains(tx.query,"cluster_id=") {t.Fatalf("multi cluster: %v %v %s",got,err,tx.query)}
 t.Log("CONFIRMED: a same-name Node in a second cluster suppresses the valid historical relation.")
}
